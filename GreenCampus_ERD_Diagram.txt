erDiagram
    users {
        int id PK
        string full_name
        string email
        string password
        string student_id
        string department
        string university
        string avatar_url
        int role_id
        timestamp created_at
    }
    items {
        bigint id PK
        bigint user_id FK
        string title
        text content
        decimal price
        string item_condition
        int category_id FK
        string status
        int quantity
        timestamp created_at
    }
    item_images {
        int id PK
        bigint item_id FK
        string image_url
        boolean is_primary
    }
    categories {
        int id PK
        string name
    }
    social_posts {
        int id PK
        bigint user_id FK
        text content
        string image_url
        string post_type
        decimal price
        boolean is_sold
        timestamp created_at
    }
    post_likes {
        int post_id PK, FK
        bigint user_id PK, FK
    }
    post_comments {
        int id PK
        int post_id FK
        bigint user_id FK
        text content
        timestamp created_at
    }
    messages {
        int id PK
        bigint sender_id FK
        bigint receiver_id FK
        text content
        bigint item_id FK
        bigint post_id FK
        boolean is_read
        timestamp created_at
    }
    pinned_items {
        int id PK
        bigint user1_id FK
        bigint user2_id FK
        bigint item_id FK
        bigint pinned_by FK
        timestamp created_at
    }
    transactions {
        bigint id PK
        bigint item_id FK
        bigint buyer_id FK
        bigint seller_id FK
        decimal amount
        string status
        timestamp created_at
    }
    ratings {
        int id PK
        bigint transaction_id FK
        bigint rater_id FK
        bigint rated_user_id FK
        tinyint rating
        text comment
        timestamp created_at
    }
    notifications {
        int id PK
        bigint user_id FK
        string type
        string title
        text body
        string link_page
        boolean is_read
        timestamp created_at
    }
    wishlists {
        int id PK
        bigint user_id FK
        bigint item_id FK
        timestamp created_at
    }
